        function selectRoute(routeId) {
            currentActiveRoute = routeId;
            const route = routesData[routeId];

            // Tab Buttons UI Update
            const btn1 = document.getElementById('btn-route-1');
            const btn2 = document.getElementById('btn-route-2');

            if (routeId === 1) {
                btn1.className = "flex-1 py-2.5 rounded-xl transition-all duration-200 bg-white text-knu-700 shadow-sm flex items-center justify-center gap-1 font-bold";
                btn2.className = "flex-1 py-2.5 rounded-xl transition-all duration-200 text-slate-600 hover:text-slate-900 flex items-center justify-center gap-1 font-medium";
            } else {
                btn2.className = "flex-1 py-2.5 rounded-xl transition-all duration-200 bg-white text-knu-700 shadow-sm flex items-center justify-center gap-1 font-bold";
                btn1.className = "flex-1 py-2.5 rounded-xl transition-all duration-200 text-slate-600 hover:text-slate-900 flex items-center justify-center gap-1 font-medium";
            }

            // Update Headers & Summary
            document.getElementById('route-bus-badge').innerText = route.badge;
            document.getElementById('route-title').innerText = route.title;
            document.getElementById('live-seats-count').innerHTML = `${route.seats} <span class="text-xs font-normal text-slate-500">/ ${route.totalSeats}석</span>`;

            const currentStop = route.stops[route.currentStopIdx];
            const nextStop = route.stops[route.currentStopIdx + 1] || route.stops[route.currentStopIdx];

            document.getElementById('current-stop-name').innerText = currentStop.name;
            document.getElementById('next-stop-name').innerText = `${nextStop.name} (${nextStop.eta})`;

            // Render Map SVG
            renderRouteMap(route);

            // Render Station List
            renderStationList(route);

            // Render Timeline History
            renderTimelineHistory(route);

            // Render Schedule
            renderScheduleTable(route);

            // Render AI Info
            renderAiInfo(route);

            lucide.createIcons();
        }

        function renderRouteMap(route) {
            const svgMap = document.getElementById('route-svg-map');
            const points = route.svgPoints;
            
            let pathD = `M ${points[0].x} ${points[0].y}`;
            for (let i = 1; i < points.length; i++) {
                pathD += ` L ${points[i].x} ${points[i].y}`;
            }

            let svgContent = `
                <!-- Background Line -->
                <path d="${pathD}" stroke="#cbd5e1" stroke-width="8" stroke-linecap="round" stroke-linejoin="round"/>
                <!-- KNU Blue Active Line -->
                <path d="${pathD}" stroke="#1b4998" stroke-width="4" stroke-linecap="round" stroke-linejoin="round" opacity="0.8"/>
            `;

            // Draw Station Points
            points.forEach((p, idx) => {
                const isCurrent = idx === route.currentStopIdx;
                const fill = isCurrent ? "#1b4998" : "#ffffff";
                const stroke = "#1b4998";

                svgContent += `
                    <circle cx="${p.x}" cy="${p.y}" r="${isCurrent ? '8' : '6'}" fill="${fill}" stroke="${stroke}" stroke-width="3"/>
                    <text x="${p.x}" y="${p.y + 22}" fill="#334155" font-size="10" font-weight="bold" text-anchor="middle">${p.label}</text>
                `;
            });

            svgMap.innerHTML = svgContent;

            // Position Live Bus Pin to current stop
            const curPt = points[route.currentStopIdx];
            const busPin = document.getElementById('live-bus-pin');
            busPin.style.left = `${(curPt.x / 400) * 100}%`;
            busPin.style.top = `${(curPt.y / 250) * 100}%`;
            document.getElementById('bus-label-pin').innerText = `달구지 ${route.badge}차`;
        }

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(tabId).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.className = 'nav-btn flex flex-col items-center text-slate-400 hover:text-slate-600';
                const textSpan = btn.querySelector('span');
                if (textSpan) textSpan.className = 'text-[10px] font-medium mt-1';
            });

            const activeNav = document.getElementById(`nav-${tabId}`);
            if (activeNav) {
                activeNav.className = 'nav-btn flex flex-col items-center text-knu-600';
                const textSpan = activeNav.querySelector('span');
                if (textSpan) textSpan.className = 'text-[10px] font-bold mt-1';
            }
        }

            secondsLeft--;
            const m = String(Math.floor(secondsLeft / 60)).padStart(2, '0');
            const s = String(secondsLeft % 60).padStart(2, '0');
            if (timerEl) timerEl.innerText = `${m}:${s}`;
        }, 1000);
